cmake_minimum_required(VERSION 3.21)
project(zcrudp_consumer LANGUAGES C)
if(ZCRUDP_SOURCE_DIR)
  add_subdirectory("${ZCRUDP_SOURCE_DIR}" zcrudp)
else()
  find_package(zcrudp CONFIG REQUIRED COMPONENTS core profiles)
endif()
add_executable(consumer main.c)
target_link_libraries(consumer PRIVATE zcrudp::profiles)
enable_testing()
add_test(NAME consumer COMMAND consumer)
