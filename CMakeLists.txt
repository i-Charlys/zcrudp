cmake_minimum_required(VERSION 3.21)
project(zcrudp LANGUAGES C)

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(ZCRUDP_BUILD_TESTS "Build CTest tests" ${PROJECT_IS_TOP_LEVEL})
option(ZCRUDP_BUILD_TOOLS "Build POSIX demo and codec benchmark" OFF)
option(ZCRUDP_BUILD_PROFILES "Build optional scalar and multipart profiles" ON)
option(ZCRUDP_WARNINGS_AS_ERRORS "Treat project compiler warnings as errors" OFF)
set(ZCRUDP_WINDOW_SIZE 64 CACHE STRING "Power-of-two window size, 2..32768")
set(ZCRUDP_MAX_CHANNELS 4 CACHE STRING "Session channel count, 1..255")
if(NOT ZCRUDP_WINDOW_SIZE MATCHES "^[0-9]+$" OR
   NOT ZCRUDP_MAX_CHANNELS MATCHES "^[0-9]+$")
  message(FATAL_ERROR "Window and channel counts must be integers")
endif()
if(ZCRUDP_WINDOW_SIZE LESS 2 OR ZCRUDP_WINDOW_SIZE GREATER 32768 OR
   ZCRUDP_MAX_CHANNELS LESS 1 OR ZCRUDP_MAX_CHANNELS GREATER 255)
  message(FATAL_ERROR "Window must be 2..32768 and channels 1..255")
endif()
math(EXPR window_remainder "${ZCRUDP_WINDOW_SIZE} & (${ZCRUDP_WINDOW_SIZE} - 1)")
if(NOT window_remainder EQUAL 0)
  message(FATAL_ERROR "Window size must be a power of two")
endif()

function(zcrudp_compile_settings target)
  target_compile_features(${target} PUBLIC c_std_11)
  set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -pedantic)
    if(ZCRUDP_WARNINGS_AS_ERRORS)
      target_compile_options(${target} PRIVATE -Werror)
    endif()
  endif()
endfunction()

add_library(zcrudp src/rudp.c)
add_library(zcrudp::core ALIAS zcrudp)
set_target_properties(zcrudp PROPERTIES EXPORT_NAME core WINDOWS_EXPORT_ALL_SYMBOLS YES)
zcrudp_compile_settings(zcrudp)
target_include_directories(zcrudp PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
# ABI-relevant settings follow consumers, including installed package users.
target_compile_definitions(zcrudp PUBLIC RUDP_WINDOW_SIZE=${ZCRUDP_WINDOW_SIZE}
  RUDP_MAX_CHANNELS=${ZCRUDP_MAX_CHANNELS})

set(zcrudp_install_targets zcrudp)
if(ZCRUDP_BUILD_PROFILES)
  add_library(zcrudp_profiles src/profiles.c)
  add_library(zcrudp::profiles ALIAS zcrudp_profiles)
  set_target_properties(zcrudp_profiles PROPERTIES EXPORT_NAME profiles WINDOWS_EXPORT_ALL_SYMBOLS YES)
  zcrudp_compile_settings(zcrudp_profiles)
  target_link_libraries(zcrudp_profiles PUBLIC zcrudp)
  list(APPEND zcrudp_install_targets zcrudp_profiles)
  install(FILES include/protocol_profiles.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
endif()

if(ZCRUDP_BUILD_TESTS)
  enable_testing()
  function(zcrudp_test target)
    zcrudp_compile_settings(${target})
    if(MSVC)
      target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${target} COMMAND ${target})
  endfunction()
  add_executable(test_configured_window tests/test_window.c)
  target_link_libraries(test_configured_window PRIVATE zcrudp)
  zcrudp_test(test_configured_window)
  add_executable(test_tfv tests/test_tfv.c)
  target_link_libraries(test_tfv PRIVATE zcrudp)
  zcrudp_test(test_tfv)
  foreach(window IN ITEMS 2 32768)
    add_executable(test_window_${window} tests/test_window.c src/rudp.c)
    target_include_directories(test_window_${window} PRIVATE include)
    target_compile_definitions(test_window_${window} PRIVATE RUDP_WINDOW_SIZE=${window})
    zcrudp_test(test_window_${window})
  endforeach()
  # These existing suites assume the default ABI in their scenarios.
  if(ZCRUDP_WINDOW_SIZE EQUAL 64 AND ZCRUDP_MAX_CHANNELS EQUAL 4)
    add_executable(test_rudp tests/test_rudp.c)
    target_link_libraries(test_rudp PRIVATE zcrudp)
    zcrudp_test(test_rudp)
    if(ZCRUDP_BUILD_PROFILES)
      add_executable(test_phase4 tests/test_phase4.c)
      target_link_libraries(test_phase4 PRIVATE zcrudp_profiles)
      zcrudp_test(test_phase4)
    endif()
  endif()
endif()

if(ZCRUDP_BUILD_TOOLS)
  if(NOT UNIX)
    message(FATAL_ERROR "The demo and benchmark currently require POSIX; disable ZCRUDP_BUILD_TOOLS")
  endif()
  if(ZCRUDP_MAX_CHANNELS LESS 2)
    message(FATAL_ERROR "The two-channel demo requires ZCRUDP_MAX_CHANNELS >= 2")
  endif()
  foreach(tool IN ITEMS demo_loss bench_rudp)
    if(tool STREQUAL "demo_loss")
      add_executable(${tool} examples/demo_loss.c)
    else()
      add_executable(${tool} bench/bench_rudp.c)
    endif()
    zcrudp_compile_settings(${tool})
    target_link_libraries(${tool} PRIVATE zcrudp)
  endforeach()
endif()

install(TARGETS ${zcrudp_install_targets} EXPORT zcrudpTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/protocol_rudp.h include/protocol_tfv.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT zcrudpTargets NAMESPACE zcrudp:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/zcrudp)
configure_package_config_file(cmake/zcrudpConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/zcrudpConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/zcrudp)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/zcrudpConfig.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/zcrudp)
