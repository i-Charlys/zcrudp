set(ZCRUDP_WINDOW_SIZE 64 CACHE STRING "Power-of-two window size, 2..16384")
set(ZCRUDP_MAX_CHANNELS 4 CACHE STRING "Session channel count, 1..255")
if(NOT ZCRUDP_WINDOW_SIZE MATCHES "^[0-9]+$" OR
   NOT ZCRUDP_MAX_CHANNELS MATCHES "^[0-9]+$")
  message(FATAL_ERROR "Window and channel counts must be integers")
endif()
if(ZCRUDP_WINDOW_SIZE LESS 2 OR ZCRUDP_WINDOW_SIZE GREATER 16384 OR
   ZCRUDP_MAX_CHANNELS LESS 1 OR ZCRUDP_MAX_CHANNELS GREATER 255)
  message(FATAL_ERROR "Window must be 2..16384 and channels 1..255")
endif()
math(EXPR window_remainder "${ZCRUDP_WINDOW_SIZE} & (${ZCRUDP_WINDOW_SIZE} - 1)")
if(NOT window_remainder EQUAL 0)
  message(FATAL_ERROR "Window size must be a power of two")
endif()

function(zcrudp_compile_settings target)
  target_compile_features(${target} PUBLIC c_std_11)
  set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -pedantic)
    if(ZCRUDP_WARNINGS_AS_ERRORS)
      target_compile_options(${target} PRIVATE -Werror)
    endif()
  endif()
endfunction()

  RUDP_WINDOW_SIZE=${ZCRUDP_WINDOW_SIZE} RUDP_MAX_CHANNELS=${ZCRUDP_MAX_CHANNELS}

function(zcrudp_module name)
  add_library(zcrudp_${name} ${ARGN})
  add_library(zcrudp::${name} ALIAS zcrudp_${name})
  set_target_properties(zcrudp_${name} PROPERTIES EXPORT_NAME ${name}
    C_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN YES
    WINDOWS_EXPORT_ALL_SYMBOLS YES)
  target_compile_definitions(zcrudp_${name} PRIVATE
    $<$<STREQUAL:$<TARGET_PROPERTY:TYPE>,SHARED_LIBRARY>:ZCRUDP_BUILDING_DLL>)
  # Installed shared modules depend on each other (session -> reliability,
  # motion -> scalar): resolve siblings relative to the library itself.
  if(APPLE)
    set_target_properties(zcrudp_${name} PROPERTIES INSTALL_RPATH "@loader_path")
  elseif(UNIX)
    set_target_properties(zcrudp_${name} PROPERTIES INSTALL_RPATH "$ORIGIN")
  endif()
  zcrudp_compile_settings(zcrudp_${name})
  target_link_libraries(zcrudp_${name} PUBLIC zcrudp_config)
  set(zcrudp_install_targets ${zcrudp_install_targets} zcrudp_${name} PARENT_SCOPE)
endfunction()

# Compatibility target: selects the enabled modules without duplicating symbols.
add_library(zcrudp INTERFACE)
add_library(zcrudp::core ALIAS zcrudp)
set_target_properties(zcrudp PROPERTIES EXPORT_NAME core)
target_link_libraries(zcrudp INTERFACE zcrudp_codec)
foreach(module IN ITEMS reliability session io crypto)
  if(TARGET zcrudp_${module})
    target_link_libraries(zcrudp INTERFACE zcrudp_${module})
  endif()
endforeach()
list(APPEND zcrudp_install_targets zcrudp)

if(ZCRUDP_BUILD_PROFILES)
  add_library(zcrudp_profiles INTERFACE)
  add_library(zcrudp::profiles ALIAS zcrudp_profiles)
  set_target_properties(zcrudp_profiles PROPERTIES EXPORT_NAME profiles)
  target_link_libraries(zcrudp_profiles INTERFACE zcrudp)
  foreach(profile IN ITEMS stream scalar motion nextgen)
    string(TOUPPER ${profile} profile_upper)
    if(ZCRUDP_BUILD_${profile_upper})
      zcrudp_module(${profile} src/${profile}.c)
      target_link_libraries(zcrudp_profiles INTERFACE zcrudp_${profile})
    endif()
  endforeach()
  if(TARGET zcrudp_stream)
    target_link_libraries(zcrudp_stream PUBLIC zcrudp_session)
  endif()
  if(TARGET zcrudp_motion)
    target_link_libraries(zcrudp_motion PUBLIC zcrudp_scalar)
  endif()
  if(TARGET zcrudp_nextgen)
    target_link_libraries(zcrudp_nextgen PUBLIC zcrudp_session zcrudp_io)
  endif()
  list(APPEND zcrudp_install_targets zcrudp_profiles)
endif()

if(ZCRUDP_BUILD_TESTS AND ZCRUDP_BUILD_RELIABILITY)
  enable_testing()
  function(zcrudp_test target)
    zcrudp_compile_settings(${target})
    if(MSVC)
      target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${target} COMMAND ${target})
  endfunction()
  if(ZCRUDP_EXTERNAL_RX_STORAGE AND ZCRUDP_BUILD_SESSION)
    add_executable(test_external_rx tests/test_external_rx.c)
    target_link_libraries(test_external_rx PRIVATE zcrudp_session)
    zcrudp_test(test_external_rx)
  endif()
  add_executable(test_configured_window tests/test_window.c)
  target_link_libraries(test_configured_window PRIVATE zcrudp)
  zcrudp_test(test_configured_window)
  add_executable(test_tfv tests/test_tfv.c)
  target_link_libraries(test_tfv PRIVATE zcrudp)
  zcrudp_test(test_tfv)
  foreach(window IN ITEMS 2 16384)
    add_executable(test_window_${window} tests/test_window.c ${ZCRUDP_RELIABILITY_SOURCES})
    target_include_directories(test_window_${window} PRIVATE include)
    target_compile_definitions(test_window_${window} PRIVATE RUDP_WINDOW_SIZE=${window})
    zcrudp_test(test_window_${window})
  endforeach()
  # These existing suites assume the default ABI in their scenarios.
  if(NOT ZCRUDP_EXTERNAL_RX_STORAGE AND ZCRUDP_BUILD_SESSION AND ZCRUDP_BUILD_IO AND ZCRUDP_WINDOW_SIZE EQUAL 64 AND ZCRUDP_MAX_CHANNELS EQUAL 4)
    add_executable(test_rudp tests/test_rudp.c)
    target_link_libraries(test_rudp PRIVATE zcrudp)
    zcrudp_test(test_rudp)
    if(TARGET zcrudp_stream AND TARGET zcrudp_scalar AND TARGET zcrudp_motion)
      add_executable(test_phase4 tests/test_phase4.c)
      target_link_libraries(test_phase4 PRIVATE zcrudp_profiles)
      zcrudp_test(test_phase4)
      if(TARGET zcrudp_crypto)
        add_executable(test_phase5 tests/test_phase5.c)
        target_link_libraries(test_phase5 PRIVATE zcrudp zcrudp_profiles)
        zcrudp_test(test_phase5)
        add_executable(test_phase6 tests/test_phase6.c tests/fuzz_rudp.c)
        target_link_libraries(test_phase6 PRIVATE zcrudp zcrudp_profiles)
        zcrudp_test(test_phase6)
        if(TARGET zcrudp_nextgen)
          add_executable(test_phase7 tests/test_phase7.c)
          target_link_libraries(test_phase7 PRIVATE zcrudp zcrudp_profiles)
          zcrudp_test(test_phase7)
          add_executable(test_phase8 tests/test_phase8.c)
          target_link_libraries(test_phase8 PRIVATE zcrudp zcrudp_profiles)
          zcrudp_test(test_phase8)
        endif()
      endif()
    endif()
  endif()
endif()

    message(FATAL_ERROR "The demo and benchmark currently require POSIX; disable ZCRUDP_BUILD_TOOLS")

  endforeach()
